        window.toggleTeacher = async function(teacherId, isAssigning) {

            if(!currentUser) return;

            const actId = document.getElementById('activity-selector').value;

            if(!actId) return;

            

            const act = activities.find(a => a.id === actId);

            if(!act) return;


            let updatedAssigned = [...(act.assignedTeachers || [])];


            if(isAssigning) {

                if(!updatedAssigned.includes(teacherId)) updatedAssigned.push(teacherId);

            } else {

                updatedAssigned = updatedAssigned.filter(id => id !== teacherId);

            }


            // Update in Firestore

            try {

                await updateDoc(doc(db, PATH_ACTIVITIES, actId), {

                    assignedTeachers: updatedAssigned

                });

            } catch(e) {

                console.error("Error updating allocation:", e);

                alert("เกิดข้อผิดพลาดในการบันทึกข้อมูล");

            }

        };

        function renderSummary() {

            const tbody = document.getElementById('summary-table-body');

            tbody.innerHTML = '';


            const summaryData = teachers.map(t => {

                const stats = getParticipationStats(t.id);

                return { name: t.name, ...stats };

            });


            // Sort by total desc, then name

            summaryData.sort((a, b) => b.total !== a.total ? b.total - a.total : a.name.localeCompare(b.name, 'th'));

            });

        }

        window.delTeacher = async function(id) {

            if(!currentUser) return;

            // No alert/confirm allowed inside Immersive based on instructions, using custom prompt logic or direct delete for simplicity.

            // Since alert() and confirm() are forbidden, we will just proceed or implement a custom modal.

            // To keep it clean in 1 file without complex modal HTML, we delete directly. 

            // *User should be careful*

            try {

                // Remove teacher from any assigned activities first

                for(let act of activities) {

                    if (act.assignedTeachers && act.assignedTeachers.includes(id)) {

                        let updated = act.assignedTeachers.filter(tId => tId !== id);

                        await updateDoc(doc(db, PATH_ACTIVITIES, act.id), { assignedTeachers: updated });

                    }

                }

                // Delete teacher record

                await deleteDoc(doc(db, PATH_TEACHERS, id));

            } catch(e) {

                console.error("Error deleting teacher:", e);

            }

        };
